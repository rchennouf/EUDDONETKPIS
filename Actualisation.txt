UPDATE #Client_situation
    SET st = CASE
        WHEN cct IS NULL THEN
            CAST(
                CAST(YEAR(GETDATE()) AS VARCHAR) + '-' +
                RIGHT('00' + CAST(MONTH(CAST(st + ' 01 2000' AS DATE)) AS VARCHAR), 2) + '-01'
            AS DATE)
        ELSE
            CAST(
                CAST(
                    CASE 
                        WHEN MONTH(CAST(st + ' 01 2000' AS DATE)) <= MONTH(cct)
                        THEN YEAR(cct) + 1
                        ELSE YEAR(cct)
                    END
                AS VARCHAR) + '-' +
                RIGHT('00' + CAST(MONTH(CAST(st + ' 01 2000' AS DATE)) AS VARCHAR), 2) + '-01'
            AS DATE)
    END;

-- Actualisation Projet duration stats
select * into #PDS from openquery([ERP.EUDONET.COM],
'SELECT  p.EvtID as projetid,
         max(cast(p.EVT01 as varchar(4000))) as projet,
         max(cast(p.EVT51 as varchar(256)))  as ic,
         max(cast(p.EVT04 as varchar(256)))  as status,
         max(cast(p.EVT02 as date))          as date_ouverture,
         max(cast(p.EVT05 as date))          as date_cloture,
         max(cast(p.EVT15 as float))         as nb_jours_prevu,
         cast(a.TPL42 as varchar(256))       as non_imputable,
         cast(a.TPL07 as varchar(256))       as nature_saisie,
         year(a.Tpl02)  as annee,
         month(a.Tpl02) as mois,
         cast(sum(datediff(minute, a.Tpl02, a.Tpl89)) / 60.0 as numeric(18,2)) as nb_heures,
         cast(u.username as varchar(256))    as detail_users
 FROM [EUDO_HOTCOM_EUDOWEB].[dbo].[Event_31] p
 LEFT JOIN [EUDO_HOTCOM_EUDOWEB].[dbo].[Template_0] a
        on a.Tpl18 = p.EvtID
       and a.Tpl02 is not null
       and a.Tpl89 > a.Tpl02
 LEFT JOIN [EUDO_HOTCOM_EUDOWEB].[dbo].[USER] u
        on u.Userid = a.TPL97
 WHERE p.EVT04 not in (''Finalisé'', ''Annulé'')
    or p.EVT05 >= dateadd(year, -2, getdate())
 GROUP BY p.EvtID,
          cast(a.TPL42 as varchar(256)),
          cast(a.TPL07 as varchar(256)),
          year(a.Tpl02), month(a.Tpl02),
          cast(u.username as varchar(256))')

truncate table [EUDDONETKPIS].[dbo].Projet_duration_stats

INSERT INTO [EUDDONETKPIS].[dbo].Projet_duration_stats
    (projetid, projet, ic, status, date_ouverture, date_cloture,
     nb_jours_prevu, non_imputable, nature_saisie, annee, mois,
     nb_heures, detail_users)
SELECT
     projetid, projet, ic, status, date_ouverture, date_cloture,
     nb_jours_prevu, non_imputable, nature_saisie, annee, mois,
     nb_heures, detail_users
FROM #PDS

drop table #PDS
